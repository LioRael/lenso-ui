  code={String.raw`import "@lenso/tokens/styles.css";
import "@lenso/ui/styles.css";
import { Button } from "@lenso/ui/button";
import { TextField } from "@lenso/ui/text-field";

export function ProjectForm() {
    return (
      <form>
        <TextField.Root>
          <TextField.Label>Project name</TextField.Label>
          <TextField.Control placeholder="Untitled" />
        </TextField.Root>
        <Button type="submit">Create project</Button>
      </form>
    );
}`}
  description="Import each component from its public subpath. Your application still owns copy, layout, validation, and submission."
  title="Compose the first form"

  <Guidance>
    <GuidanceBlock title="What the library owns">
      <ul>
        <li>Foundation components provide reusable behavior, accessibility, and visual states.</li>
        <li>Semantic tokens keep color, type, spacing, and interaction feedback coherent.</li>
        <li>Use <code>ThemeScope</code> when a subtree needs an explicit Light or Dark boundary.</li>
      </ul>
    </GuidanceBlock>
    <GuidanceBlock title="What your application owns">
      <ul>
        <li>Keep product data, navigation, persistence, permissions, and interface copy outside the library.</li>
        <li>Browse a component page before composing its compound parts.</li>
        <li>Choose registry source only when you need to change component structure.</li>
      </ul>
    </GuidanceBlock>
  </Guidance>
