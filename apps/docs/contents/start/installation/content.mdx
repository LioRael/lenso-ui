import { Guidance, GuidanceBlock } from "@/components/docs/guidance";

  <Guidance>
    <GuidanceBlock title="Use the package by default">
      Install <code>@lenso/ui</code> when you want versioned upgrades and precompiled StyleX output.
      Import <code>@lenso/tokens/styles.css</code> and <code>@lenso/ui/styles.css</code> once from the
      application shell.
    </GuidanceBlock>
    <GuidanceBlock title="Copy source only when you need it">
      Install a registry item when your application must change component structure or own a Recipe.
      Registry files become application code and do not receive package updates automatically.
    </GuidanceBlock>
  </Guidance>

<CodeExample
  code={String.raw`pnpm add @lenso/ui @lenso/tokens @base-ui/react react react-dom`}
  description="Install the React and Base UI peers."
  language="bash"
  title="Install the package channel"
/>

<CodeExample
  code={String.raw`import "@lenso/tokens/styles.css";
import "@lenso/ui/styles.css";`}
  description="Load each generated stylesheet once at the application entry or root layout."
  title="Load the styles"
/>

<CodeExample
  code={String.raw`pnpm dlx shadcn@latest add https://ui.lenso.dev/r/setup.json
pnpm dlx shadcn@latest add https://ui.lenso.dev/r/button.json`}
  description="The setup item adds the shared token bridge. Every component item is copied into your application as editable source."
  language="bash"
  title="Install registry source"
/>

  <Guidance>
    <GuidanceBlock title="Before copying registry source">
      <ul>
        <li>Configure a compatible StyleX compiler in your application.</li>
        <li>Keep the generated token CSS import in the application shell.</li>
        <li>Use a versioned registry URL when a build must resolve the same source later.</li>
      </ul>
    </GuidanceBlock>
    <GuidanceBlock title="Optional packages">
      <ul>
        <li>Install <code>@lenso/primitives</code> directly only when you need the headless Sidebar primitive.</li>
        <li><code>@lenso/ui/preflight.css</code> is optional and is not a reset or font package.</li>
      </ul>
    </GuidanceBlock>
  </Guidance>
