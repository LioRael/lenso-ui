<Playground id="menu" />

  <Guidance>
    <GuidanceBlock title="Choose Menu when">
      <ul>
        <li>Use Item for actions and LinkItem for navigation.</li>
        <li>Keep labels concise and group destructive actions last.</li>
        <li>Use <code>layout="stacked"</code> only when one short context line distinguishes choices, such as workspaces with similar names. Omit repeated tags and capability lists.</li>
        <li>Use RadioItem for the selected workspace and keep icon, copy, and check aligned on one row.</li>
      </ul>
    </GuidanceBlock>
    <GuidanceBlock title="Accessibility">
      <ul>
        <li>Base UI owns roles, roving focus, typeahead, and nested keyboard navigation.</li>
        <li>Escape closes the active surface and restores focus.</li>
      </ul>
    </GuidanceBlock>
  </Guidance>

  <CodeExample
    code={String.raw`import { Menu } from "@lenso/ui/menu";

<Menu.Root>
    <Menu.ControlTrigger>Open menu</Menu.ControlTrigger>
    <Menu.Portal>
      <Menu.Positioner>
        <Menu.Popup>
          <Menu.LinkItem href="/issue/14">Open issue</Menu.LinkItem>
          <Menu.SubmenuRoot>
            <Menu.SubmenuTrigger>Create related</Menu.SubmenuTrigger>
            <Menu.Portal>
              <Menu.Positioner side="right">
                <Menu.Popup submenu>...</Menu.Popup>
              </Menu.Positioner>
            </Menu.Portal>
          </Menu.SubmenuRoot>
        </Menu.Popup>
      </Menu.Positioner>
    </Menu.Portal>
</Menu.Root>;`}
    description="Actions, navigation, and selection share the same overlay surface; product context belongs in concise labels and descriptions."
  />

    description="Place the RadioGroup inside Menu.Popup. Use the description only when it helps distinguish a choice; the selected check and concise label carry the main decision."
