<Playground id="icon-button" />

  <Guidance>
    <GuidanceBlock title="Choose Icon Button when">
      <ul>
        <li>The action is common, its icon is widely understood, and nearby text supplies context.</li>
        <li>Use Button when the label carries meaning or the action needs stronger emphasis.</li>
        <li>Keep the icon visually smaller than its interaction target.</li>
      </ul>
    </GuidanceBlock>
    <GuidanceBlock title="Naming and feedback">
      <ul>
        <li>Always provide an accessible name with <code>aria-label</code>.</li>
        <li>Use selected state only for a persistent toggle; use pressed feedback for ordinary actions.</li>
        <li>Add a Tooltip when the icon may be unfamiliar, but do not use it as the accessible name.</li>
      </ul>
    </GuidanceBlock>
  </Guidance>

  <CodeExample
    code={String.raw`import { IconButton } from "@lenso/ui/icon-button"
import { PlusIcon } from "lucide-react"

<IconButton aria-label="Create issue">
  <PlusIcon />
</IconButton>`}
    description="The component keeps icon alignment and target size stable across visual variants."
  />
