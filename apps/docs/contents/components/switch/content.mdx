<Playground id="switch" />

  <Guidance>
    <GuidanceBlock title="Choose Switch when">
      <ul>
        <li>The setting is binary and takes effect as soon as it changes.</li>
        <li>Use Checkbox when the value belongs to a form that is submitted later.</li>
        <li>
          Use the control-only layout when a visible label outside the Switch names it, such as a
          Settings Row label.
        </li>
      </ul>
    </GuidanceBlock>
    <GuidanceBlock title="Labels and feedback">
      <ul>
        <li>Use a label that describes the setting, not the action: “Notifications,” not “Enable.”</li>
        <li>Keep compact switches control-only and connect them to a visible external label.</li>
        <li>Announce a failure if the application cannot persist the new state.</li>
      </ul>
    </GuidanceBlock>
  </Guidance>

  <CodeExample
    code={String.raw`import { Switch } from "@lenso/ui/switch"

<Switch.Root aria-label="Notifications" defaultChecked layout="control-only">
  <Switch.Thumb />
</Switch.Root>`}
    description="Root owns switch semantics; layout controls whether its visible label is internal or supplied by the surrounding pattern."
  />
