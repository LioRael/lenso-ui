<Playground id="checkbox" />

  <Guidance>
    <GuidanceBlock title="Choose Checkbox when">
      <ul>
        <li>Each option can be selected independently of the others.</li>
        <li>Use the indeterminate state only to summarize a partially selected child group.</li>
        <li>Use Switch instead when changing the value should take effect immediately.</li>
      </ul>
    </GuidanceBlock>
    <GuidanceBlock title="Labels and state">
      <ul>
        <li>Keep a visible label beside the control; do not rely on surrounding copy.</li>
        <li>Base UI exposes checked, mixed, disabled, and keyboard-focus states.</li>
        <li>Explain why an option is unavailable instead of leaving a disabled control unexplained.</li>
      </ul>
    </GuidanceBlock>
  </Guidance>

  <CodeExample
    code={String.raw`import { Checkbox } from "@lenso/ui/checkbox";

<Checkbox.Root defaultChecked>
    <Checkbox.Indicator />
    <Checkbox.Label>Include completed issues</Checkbox.Label>
</Checkbox.Root>;`}
    description="Root owns checkbox behavior while Indicator and Label keep the visible treatment composable."
  />
