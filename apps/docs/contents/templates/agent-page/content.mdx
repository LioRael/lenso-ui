  <Guidance>
    <GuidanceBlock title="Compose the prompt workflow">
      <ul>
        <li>Your application passes the draft, change handler, submit handler, and action controls.</li>
        <li>Plain Enter inserts a newline by default; Control or Command + Enter submits.</li>
        <li>
          Existing Enter-to-send products may choose submitShortcut=&quot;enter&quot;. Shift+Enter still
          inserts a newline and IME composition never submits.
        </li>
        <li>Autosizing follows text reflow and stops at the row limit you choose.</li>
        <li>Use the Recipe's surfaceXstyle and part-level xstyle seams to match the owning product's composer surface and density; keep its keyboard and autosize behavior.</li>
      </ul>
    </GuidanceBlock>
    <GuidanceBlock title="Keep product behavior outside the Recipe">
      <ul>
        <li>The Recipe does not create or persist Agent sessions.</li>
        <li>Skills, attachments, tools, and their permission flows remain product-owned slots.</li>
        <li>The application decides whether an empty draft is valid; the Recipe does not disable it.</li>
      </ul>
    </GuidanceBlock>
  </Guidance>

<CodeExample
  code={String.raw`import { useState } from "react"
import { Button } from "@lenso/ui/button"
import { PromptComposer } from "@/components/lenso/recipes/prompt-composer"

export function AgentComposer() {
  const [draft, setDraft] = useState("")

  return (
    <PromptComposer.Root
      value={draft}
      onValueChange={setDraft}
      onSubmit={(event) => {
        event.preventDefault()
        // Validate and dispatch through the application session here.
      }}
    >
      <PromptComposer.Input aria-label="Message" placeholder="Describe the work…" />
      <PromptComposer.Toolbar>
        <span>Control or Command + Enter to submit</span>
        <PromptComposer.Actions>
          <Button type="submit">Send</Button>
        </PromptComposer.Actions>
      </PromptComposer.Toolbar>
    </PromptComposer.Root>
  )
}`}
  description="PromptComposer is a presentational and keyboard seam. Session lifecycle and product actions stay outside it."
/>
