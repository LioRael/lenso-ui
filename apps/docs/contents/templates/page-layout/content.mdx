  <Guidance>
    <GuidanceBlock title="Structure the shell">
      <ul>
        <li>
          Keep product navigation inside Sidebar and pass product-owned content to the main landmark.
        </li>
        <li>Use the same shell geometry for dashboard, table, list, and settings workflows.</li>
        <li>
          In the settings context, render the Settings Page template as the main landmark instead of
          leaving the work surface empty.
        </li>
        <li>Keep low-frequency environment actions in the bottom utility strip.</li>
      </ul>
    </GuidanceBlock>
    <GuidanceBlock title="Adapt the navigation on small screens">
      <ul>
        <li>The desktop reference uses a 244px sidebar around an inset primary surface.</li>
        <li>
          At narrow widths, let Sidebar.Root own disclosure while your application chooses overlay or
          compact navigation.
        </li>
        <li>
          Preserve the main landmark, focus order, and current route when the sidebar changes
          presentation.
        </li>
      </ul>
    </GuidanceBlock>
  </Guidance>

  <CodeExample
    code={String.raw`import * as stylex from "@stylexjs/stylex"
import { BotIcon, HistoryIcon } from "lucide-react"
import { Button } from "@lenso/ui/button"
import { IconButton } from "@lenso/ui/icon-button"
import { Sidebar } from "@lenso/ui/sidebar"
import { ThemeScope } from "@lenso/ui/theme-scope"
import { styles } from "./page-layout.stylex"

export function PageLayout({ children }: { children: React.ReactNode }) {
  return (
    <ThemeScope theme="system">
      <Sidebar.Group xstyle={styles.shell}>
        <Sidebar.Root defaultOpen id="app-sidebar" xstyle={styles.sidebar}>
          <Sidebar.Panel aria-label="Workspace navigation">
            <Sidebar.Header>{/* workspace and actions */}</Sidebar.Header>
            <Sidebar.Content>{/* product navigation */}</Sidebar.Content>
          </Sidebar.Panel>
        </Sidebar.Root>

        <main {...stylex.props(styles.main)}>{children}</main>

        <footer aria-label="Application utilities" {...stylex.props(styles.utilities)}>
          <Button variant="ghost"><BotIcon aria-hidden />Agent</Button>
          <IconButton aria-label="Open chat history" variant="ghost">
            <HistoryIcon />
          </IconButton>
        </footer>
      </Sidebar.Group>
    </ThemeScope>

)
}`}
description="The template composes stable Lenso UI parts; routing, navigation data, and page content remain application-owned."
/>
